% --- Operators ---------------------------------------------------------------
\providecommand{\dalembert}{\Box}                 % d'Alembertian
\providecommand{\Tr}{\operatorname{Tr}}           % trace
\providecommand{\Real}{\operatorname{Re}}         % real part
\providecommand{\Imag}{\operatorname{Im}}         % imaginary part

% --- Common symbols of this paper -------------------------------------------
\providecommand{\PLKG}{\text{PLKG}}               % Phase-Locked Klein-Gordon label
\providecommand{\sync}{\text{sync}}               % synchronization subscript
\providecommand{\intpot}{V_{\mathrm{int}}}        % interaction potential
\providecommand{\meff}{m_{\mathrm{eff}}}          % effective mass

% --- Convenience -------------------------------------------------------------
\providecommand{\eqlabel}[1]{\label{eq:#1}}       % uniform equation labelling

% --- PRD / APS citations (revtex4-2 + natbib) --------------------------------

% reference number must appear inline (e.g., ``Refs.~\onlinecite{Hu2000}'').
\providecommand{\Refcite}[1]{Ref.~\onlinecite{#1}}
\providecommand{\Refscite}[1]{Refs.~\onlinecite{#1}}
